    src/native.cpp src/blueprint.cpp src/texture.cpp src/image.cpp src/hdr.cpp src/codec.cpp src/mesh.cpp src/material_io.cpp src/material.cpp)

    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/third_party>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)

    add_compile_options(/utf-8 /EHsc)
    target_compile_options(cyuediff_core PRIVATE /W4 /utf-8 /EHsc)
    target_compile_definitions(cyuediff_core PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(cyuediff_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    target_link_libraries(cyuediff_core PRIVATE advapi32)
elseif(CMAKE_DL_LIBS)
    target_link_libraries(cyuediff_core PRIVATE ${CMAKE_DL_LIBS})
endif()
if(CYUEDIFF_BUILD_GIT)

    add_library(CyUEDiff::Git ALIAS cyuediff_git)
    target_link_libraries(cyuediff_git PUBLIC CyUEDiff::Core)
    set_target_properties(cyuediff_git PROPERTIES EXPORT_NAME Git POSITION_INDEPENDENT_CODE ON)
    if(MSVC)
        target_compile_options(cyuediff_git PRIVATE /W4 /utf-8 /EHsc)
    else()
        target_compile_options(cyuediff_git PRIVATE -Wall -Wextra -Wpedantic)
        find_package(Threads REQUIRED)
        target_link_libraries(cyuediff_git PRIVATE Threads::Threads)
    endif()
    install(TARGETS cyuediff_git EXPORT CyUEDiffTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})

    endif()

    if(NOT TARGET CyUI::Diff OR NOT TARGET CyUI::Nodes OR NOT TARGET CyUI::Images OR NOT TARGET CyUI::Viewport)
        if(NOT EXISTS "${CYUEDIFF_CYUI_SOURCE_DIR}/CMakeLists.txt")

        endif()

        set(CYUI_BUILD_AUDIO OFF)
        set(CYUI_BUILD_GALLERY OFF)
