cmake_minimum_required(VERSION 3.20)
project(CyUEDiff VERSION 0.12.0 LANGUAGES CXX)
include(GNUInstallDirs)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    set(CYUEDIFF_TOP_LEVEL ON)
else()
    set(CYUEDIFF_TOP_LEVEL OFF)
endif()
option(BUILD_TESTING "Build tests" ${CYUEDIFF_TOP_LEVEL})
include(CTest)
option(CYUEDIFF_BUILD_CLI "Build the command line application" ${CYUEDIFF_TOP_LEVEL})
option(CYUEDIFF_BUILD_GIT "Build the separate local Git/LFS adapter" ${CYUEDIFF_TOP_LEVEL})
option(CYUEDIFF_BUILD_DESKTOP "Build the Windows desktop application" OFF)
if(CYUEDIFF_BUILD_DESKTOP)
    if(NOT WIN32)
        message(FATAL_ERROR "The desktop host currently requires Windows")
    endif()
    set(CYUEDIFF_BUILD_UI ON)
endif()
option(CYUEDIFF_BUILD_UI "Build the separate CyUI view library" OFF)
set(CYUEDIFF_CYUI_SOURCE_DIR "" CACHE PATH "CyUI source checkout")
add_library(cyuediff_core STATIC
    src/archive.cpp src/decoders.cpp src/descriptor.cpp src/package.cpp src/project.cpp src/catalog.cpp src/diff.cpp src/visual_diff.cpp
    src/native.cpp src/blueprint.cpp src/texture.cpp src/image.cpp src/codec.cpp src/mesh.cpp src/material_io.cpp src/material.cpp)
add_library(CyUEDiff::Core ALIAS cyuediff_core)
target_compile_features(cyuediff_core PUBLIC cxx_std_17)
target_include_directories(cyuediff_core PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/third_party>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(cyuediff_core PROPERTIES POSITION_INDEPENDENT_CODE ON EXPORT_NAME Core)
if(MSVC)
    add_compile_options(/utf-8 /EHsc)
    target_compile_options(cyuediff_core PRIVATE /W4 /utf-8 /EHsc)
    target_compile_definitions(cyuediff_core PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(cyuediff_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    target_link_libraries(cyuediff_core PRIVATE advapi32)
endif()
if(CYUEDIFF_BUILD_GIT)
    add_library(cyuediff_git STATIC src/git.cpp src/git_process.cpp src/git_project.cpp)
    add_library(CyUEDiff::Git ALIAS cyuediff_git)
    target_link_libraries(cyuediff_git PUBLIC CyUEDiff::Core)
    set_target_properties(cyuediff_git PROPERTIES EXPORT_NAME Git POSITION_INDEPENDENT_CODE ON)
    if(MSVC)
        target_compile_options(cyuediff_git PRIVATE /W4 /utf-8 /EHsc)
    else()
        target_compile_options(cyuediff_git PRIVATE -Wall -Wextra -Wpedantic)
        find_package(Threads REQUIRED)
        target_link_libraries(cyuediff_git PRIVATE Threads::Threads)
    endif()
    install(TARGETS cyuediff_git EXPORT CyUEDiffTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
    install(FILES include/cyuediff/git.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/cyuediff)
endif()
if(CYUEDIFF_BUILD_CLI)
    add_executable(cyuediff apps/cli/main.cpp)
    target_link_libraries(cyuediff PRIVATE CyUEDiff::Core)
    target_compile_definitions(cyuediff PRIVATE CYUEDIFF_VERSION="${PROJECT_VERSION}")
    if(CYUEDIFF_BUILD_GIT)
        target_link_libraries(cyuediff PRIVATE CyUEDiff::Git)
        target_compile_definitions(cyuediff PRIVATE CYUEDIFF_HAS_GIT=1)
    endif()
endif()
if(CYUEDIFF_BUILD_UI)
    if(TARGET CyUI::Core AND (NOT TARGET CyUI::Diff OR NOT TARGET CyUI::Nodes OR NOT TARGET CyUI::Images OR NOT TARGET CyUI::Viewport))
        message(FATAL_ERROR "Configure the host's CyUI with CYUI_MODULES Diff;Nodes;Images;Viewport before adding CyUEDiff")
    endif()
    if(NOT TARGET CyUI::Diff OR NOT TARGET CyUI::Nodes OR NOT TARGET CyUI::Images OR NOT TARGET CyUI::Viewport)
        if(NOT EXISTS "${CYUEDIFF_CYUI_SOURCE_DIR}/CMakeLists.txt")
            message(FATAL_ERROR "Set CYUEDIFF_CYUI_SOURCE_DIR to a CyUI checkout")
        endif()
        set(CYUI_BUILD_DEMO OFF)
        set(CYUI_BUILD_AUDIO OFF)
        set(CYUI_BUILD_GALLERY OFF)
        set(CYUI_MODULES "Diff;Nodes;Images;Viewport")
        add_subdirectory("${CYUEDIFF_CYUI_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/cyui")
    endif()
    add_library(cyuediff_ui STATIC ui/property_view.cpp ui/catalog_view.cpp ui/asset_view.cpp ui/blueprint_view.cpp ui/texture_view.cpp ui/mesh_view.cpp ui/material_view.cpp)
    add_library(CyUEDiff::UI ALIAS cyuediff_ui)
    target_include_directories(cyuediff_ui PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
    target_link_libraries(cyuediff_ui PUBLIC CyUEDiff::Core CyUI::Diff CyUI::Nodes CyUI::Images CyUI::Viewport)
endif()
if(CYUEDIFF_BUILD_DESKTOP)
    set(CYUEDIFF_IMGUI_BACKENDS_DIR "${CYUEDIFF_CYUI_SOURCE_DIR}/third_party/imgui/backends" CACHE PATH "Matching ImGui Win32/DX11 backend sources")
    if(NOT EXISTS "${CYUEDIFF_IMGUI_BACKENDS_DIR}/imgui_impl_win32.cpp" OR NOT EXISTS "${CYUEDIFF_IMGUI_BACKENDS_DIR}/imgui_impl_dx11.cpp")
        message(FATAL_ERROR "Set CYUEDIFF_IMGUI_BACKENDS_DIR to the backend sources matching the host's ImGui")
    endif()
    add_library(cyuediff_desktop_model STATIC apps/desktop/document.cpp apps/desktop/browser.cpp)
    target_link_libraries(cyuediff_desktop_model PUBLIC CyUEDiff::Core)
    target_include_directories(cyuediff_desktop_model PUBLIC apps/desktop)
    if(CYUEDIFF_BUILD_GIT)
        target_link_libraries(cyuediff_desktop_model PUBLIC CyUEDiff::Git)
        target_compile_definitions(cyuediff_desktop_model PUBLIC CYUEDIFF_HAS_GIT=1)
    endif()
    add_executable(cyuediff_desktop WIN32 apps/desktop/main.cpp apps/desktop/platform_win32.cpp
        "${CYUEDIFF_IMGUI_BACKENDS_DIR}/imgui_impl_win32.cpp"
        "${CYUEDIFF_IMGUI_BACKENDS_DIR}/imgui_impl_dx11.cpp")
    target_include_directories(cyuediff_desktop PRIVATE "${CYUEDIFF_IMGUI_BACKENDS_DIR}")
    target_compile_definitions(cyuediff_desktop PRIVATE NOMINMAX UNICODE _UNICODE)
    target_link_libraries(cyuediff_desktop PRIVATE cyuediff_desktop_model CyUEDiff::UI d3d11 dxgi comdlg32 shell32)
endif()
if(BUILD_TESTING)
    add_executable(cyuediff_visual_diff_tests tests/visual_diff_tests.cpp)
    target_link_libraries(cyuediff_visual_diff_tests PRIVATE CyUEDiff::Core)
    add_test(NAME visual_diff COMMAND cyuediff_visual_diff_tests)
    add_executable(cyuediff_tests tests/core_tests.cpp)
    target_link_libraries(cyuediff_tests PRIVATE CyUEDiff::Core)
    add_test(NAME core COMMAND cyuediff_tests)
    add_executable(cyuediff_catalog_tests tests/catalog_tests.cpp)
    target_link_libraries(cyuediff_catalog_tests PRIVATE CyUEDiff::Core)
    add_test(NAME catalog COMMAND cyuediff_catalog_tests)
    add_executable(cyuediff_decoder_tests tests/decoder_tests.cpp)
    target_link_libraries(cyuediff_decoder_tests PRIVATE CyUEDiff::Core)
    add_test(NAME decoders COMMAND cyuediff_decoder_tests)
    add_executable(cyuediff_descriptor_tests tests/descriptor_tests.cpp)
    target_link_libraries(cyuediff_descriptor_tests PRIVATE CyUEDiff::Core)
    add_test(NAME descriptor COMMAND cyuediff_descriptor_tests)
    add_executable(cyuediff_mesh_tests tests/mesh_tests.cpp)
    target_link_libraries(cyuediff_mesh_tests PRIVATE CyUEDiff::Core)
    add_test(NAME mesh COMMAND cyuediff_mesh_tests)
    add_executable(cyuediff_texture_tests tests/texture_tests.cpp)
    target_link_libraries(cyuediff_texture_tests PRIVATE CyUEDiff::Core)
    add_test(NAME texture COMMAND cyuediff_texture_tests)
    add_executable(cyuediff_material_tests tests/material_tests.cpp)
    target_link_libraries(cyuediff_material_tests PRIVATE CyUEDiff::Core)
    add_test(NAME material COMMAND cyuediff_material_tests)
    add_executable(cyuediff_blueprint_tests tests/blueprint_tests.cpp)
    target_link_libraries(cyuediff_blueprint_tests PRIVATE CyUEDiff::Core)
    add_test(NAME blueprint COMMAND cyuediff_blueprint_tests)
    if(CYUEDIFF_BUILD_GIT)
        add_executable(cyuediff_git_tests tests/git_tests.cpp)
        target_link_libraries(cyuediff_git_tests PRIVATE CyUEDiff::Git)
        add_executable(cyuediff_git_project_tests tests/git_project_tests.cpp)
        target_link_libraries(cyuediff_git_project_tests PRIVATE CyUEDiff::Git)
        find_package(Git 2.45 QUIET)
        if(GIT_FOUND)
            add_test(NAME git COMMAND cyuediff_git_tests --git "${GIT_EXECUTABLE}")
            add_test(NAME git_project COMMAND cyuediff_git_project_tests --git "${GIT_EXECUTABLE}")
        else()
            message(STATUS "Git adapter built; integration tests require Git 2.45 or later")
        endif()
    endif()
    if(CYUEDIFF_BUILD_DESKTOP)
        add_executable(cyuediff_browser_tests tests/browser_tests.cpp)
        target_link_libraries(cyuediff_browser_tests PRIVATE cyuediff_desktop_model)
        add_test(NAME desktop_browser COMMAND cyuediff_browser_tests)
        set_tests_properties(desktop_browser PROPERTIES TIMEOUT 45)
        add_executable(cyuediff_desktop_tests tests/desktop_tests.cpp)
        target_link_libraries(cyuediff_desktop_tests PRIVATE cyuediff_desktop_model)
        add_test(NAME desktop_model COMMAND cyuediff_desktop_tests)
        add_test(NAME desktop_renderer COMMAND cyuediff_desktop --smoke)
        set_tests_properties(desktop_model desktop_renderer PROPERTIES TIMEOUT 45)
        if(CYUEDIFF_BUILD_GIT AND GIT_FOUND)
            add_executable(cyuediff_desktop_git_tests tests/desktop_git_tests.cpp)
            target_link_libraries(cyuediff_desktop_git_tests PRIVATE cyuediff_desktop_model)
            add_test(NAME desktop_git COMMAND cyuediff_desktop_git_tests --git "${GIT_EXECUTABLE}")
            set_tests_properties(desktop_git PROPERTIES TIMEOUT 120)
        endif()
    endif()
    if(CYUEDIFF_BUILD_UI)
        add_executable(cyuediff_ui_smoke tests/ui_smoke.cpp)
        target_link_libraries(cyuediff_ui_smoke PRIVATE CyUEDiff::UI)
        add_test(NAME ui_smoke COMMAND cyuediff_ui_smoke)
    endif()
endif()
install(TARGETS cyuediff_core EXPORT CyUEDiffTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/cyuediff DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
    PATTERN "property_view.hpp" EXCLUDE PATTERN "git.hpp" EXCLUDE PATTERN "catalog_view.hpp" EXCLUDE PATTERN "asset_view.hpp" EXCLUDE PATTERN "blueprint_view.hpp" EXCLUDE PATTERN "texture_view.hpp" EXCLUDE PATTERN "mesh_view.hpp" EXCLUDE PATTERN "material_view.hpp" EXCLUDE)
install(DIRECTORY third_party/nlohmann DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} FILES_MATCHING PATTERN "*.hpp")
install(EXPORT CyUEDiffTargets NAMESPACE CyUEDiff:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CyUEDiff)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CyUEDiffConfig.cmake" "include(\"\${CMAKE_CURRENT_LIST_DIR}/CyUEDiffTargets.cmake\")\n")
if(CYUEDIFF_BUILD_GIT AND NOT MSVC)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CyUEDiffConfig.cmake" "include(CMakeFindDependencyMacro)\nfind_dependency(Threads)\ninclude(\"\${CMAKE_CURRENT_LIST_DIR}/CyUEDiffTargets.cmake\")\n")
endif()
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/CyUEDiffConfig.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CyUEDiff)

install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION ${CMAKE_INSTALL_DATADIR}/CyUEDiff/licenses/nlohmann)

install(FILES third_party/stb/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/CyUEDiff/licenses/stb)

# Optional locally built codec; never part of the installed Core SDK.
set(CYUEDIFF_OODLE_SDK_DIR "" CACHE PATH "Host-owned Oodle Data SDK (optional, Windows x64)")
if(CYUEDIFF_OODLE_SDK_DIR)
    if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The local Oodle adapter currently requires Windows x64/MSVC")
    endif()
    if(NOT EXISTS "${CYUEDIFF_OODLE_SDK_DIR}/include/oodle2.h" OR
       NOT EXISTS "${CYUEDIFF_OODLE_SDK_DIR}/lib/Win64/oo2core_win64.lib")
        message(FATAL_ERROR "CYUEDIFF_OODLE_SDK_DIR must point to a host-provided Oodle Data SDK")
    endif()
    add_library(cyuediff_oodle MODULE codecs/oodle_adapter.cpp)
    target_include_directories(cyuediff_oodle PRIVATE "${CYUEDIFF_OODLE_SDK_DIR}/include")
    target_link_libraries(cyuediff_oodle PRIVATE "${CYUEDIFF_OODLE_SDK_DIR}/lib/Win64/oo2core_win64.lib")
    target_compile_features(cyuediff_oodle PRIVATE cxx_std_17)
    set_target_properties(cyuediff_oodle PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/engine-codecs" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/engine-codecs")
endif()

install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/CyUEDiff)
