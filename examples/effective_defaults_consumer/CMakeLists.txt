cmake_minimum_required(VERSION 3.16)
project(EffectiveDefaultsConsumer LANGUAGES CXX)
find_package(CyUEDiff 0.200.0 EXACT CONFIG REQUIRED)
add_executable(effective_defaults_consumer main.cpp)
target_link_libraries(effective_defaults_consumer PRIVATE CyUEDiff::Core)
target_compile_features(effective_defaults_consumer PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(effective_defaults_consumer PRIVATE /utf-8 /EHsc)
endif()
