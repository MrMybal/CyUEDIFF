cmake_minimum_required(VERSION 3.20)
project(CyUEDiffGitConsumer LANGUAGES CXX)
find_package(CyUEDiff CONFIG REQUIRED)
if(NOT TARGET CyUEDiff::Git)
    message(FATAL_ERROR "Install CyUEDiff with CYUEDIFF_BUILD_GIT=ON first")
endif()
add_executable(git_compare main.cpp)
target_link_libraries(git_compare PRIVATE CyUEDiff::Git)
if(MSVC)
    target_compile_options(git_compare PRIVATE /EHsc /utf-8)
endif()
